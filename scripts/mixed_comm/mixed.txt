LOAD_PLUGIN KVCAN

TOUT_TX := 5000
SZ_BUFF := 64
CANIF   := vcan0
TX_ID   := 0x100

carray [= 1000:>H'11223344'

KVCAN.CONFIG $CFG_COMMON

recval ?= UART.CYCLIC $carray &
sndval ?= KVCAN.CMD < &

DELAY 5 sec
