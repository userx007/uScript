LOAD_PLUGIN UART
LOAD_PLUGIN KCAN

TOUT_RX := 5000
TOUT_TX := 5000
SZ_BUFF := 64
CANIF   := vcan0
TX_ID   := 0x100
CFG_COMMON := i=$CANIF r=$TOUT_RX w=$TOUT_TX s=$SZ_BUFF x=$TX_ID

carray [= 1000:>H'11223344'

KCAN.CONFIG $CFG_COMMON

recval ?= UART.CYCLIC $carray &
sndval ?= KCAN.CMD < &

DELAY 15 sec
