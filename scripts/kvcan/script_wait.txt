LOAD_PLUGIN KVCAN
LOAD_PLUGIN SHELL

TOUT_RX := 10000
TOUT_TX := 2000
SZ_BUFF := 64
CANIF   := vcan0
TX_ID   := 0x100

CFG_COMMON  := i=$CANIF r=$TOUT_RX w=$TOUT_TX s=$SZ_BUFF 

KVCAN.CONFIG $CFG_COMMON x=$TX_ID

ITEM := 255:1:1 0:1:1

A ?= BITSTREAM $ITEM
PRINT A=$A 

A_RBI ?= BITSTREAM $ITEM | REVERSE_BIT
PRINT A_RBI=$A_RBI 

A_RBY ?= BITSTREAM $ITEM | REVERSE_BYTE
PRINT A_RBY=$A_RBY

ITEM_1 := 32:7:1 0:1:1

B ?= BYTESTREAM $ITEM_1
PRINT B=$B

B_RBI ?= BYTESTREAM $ITEM_1 | REVERSE_BIT
PRINT B_RBI=$B_RBI 

B_RBY ?= BYTESTREAM $ITEM_1 | REVERSE_BYTE
