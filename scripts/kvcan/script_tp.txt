LOAD_PLUGIN KVCAN

TOUT_RX := 5000
TOUT_TX := 5000
SZ_BUFF := 64
CANIF   := vcan0
TX_ID   := 0x100
RX_ID   := 0x101



CFG_COMMON      := i=$CANIF r=$TOUT_RX w=$TOUT_TX s=$SZ_BUFF 
CFG_ISOTP       := x=0x100 y=0x101 t=isotp
CFG_j1939RTSCTS := x=0x200 y=0x201 t=j1939
CFG_j1939BAM    := x=0x300 y=0x301 t=j1939 bam=TRUE
CFG_j1939NMEA   := x=0x400 y=0x401 t=nmea2000
CFG_CANOPEN     := x=0x500 y=0x501 t=canopen

#CFG_TP := x=$TX_ID y=$RX_ID
CFG_TP := $CFG_ISOTP
#CFG_TP := $CFG_j1939RTSCTS
#CFG_TP := $CFG_j1939BAM
#CFG_TP := $CFG_j1939NMEA
#CFG_TP := $CFG_CANOPEN

KVCAN.CONFIG $CFG_COMMON $CFG_TP

KVCAN.CMD > H'00112233' | H'00112233'
KVCAN.CMD > H'AABBCCDD112233445566778899001122' | H'AABBCCDD112233445566778899001122'

array0 [= 0x11, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77, 0x88, 0x99
array1 [= 0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF, 0xEE, 0xDD, 0xCC
var3 ?= 11223344

i  ?=  REPEAT  endloop2 $array0.SIZE
    val0 ?= MATH $array0.$i | HEX_32_LE
    val1 ?= MATH $array1.$i | HEX_32_LE

    KVCAN.CMD > H'$val0$val1$var3' | H'$val0$val1$var3'
    DELAY 1000 ms
END_REPEAT  endloop2


KVCAN.SCRIPT skcan.txt
