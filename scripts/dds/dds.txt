LOAD_PLUGIN DDS
LOAD_PLUGIN DDS_TYPED

received ?= ""

#DDS.INFO
DDS_TYPED.INFO

DDS:1.CONFIG d=99 pid=1 n=loopback_pub
DDS:2.CONFIG d=99 pid=2 n=loopback_sub

DDS:1.CMD > LIST
DELAY 1 sec
DDS:2.CMD > SUBSCRIBE selftest/ping
received ?= DDS:2.CMD < &

# Give SPDP+SEDP a moment to discover each other and match the topic
# before publishing — see Gotchas for why this delay matters.
DELAY 1000 ms

DDS:1.CMD > PUBLISH selftest/ping hello2you

REPEAT wait_ping UNTIL EVAL $received != "" | STR
DELAY 100 ms
END_REPEAT wait_ping

IF EVAL $received == "hello" | STR GOTO pass
PRINT SELF-TEST FAILED: expected 'hello', got '$received'
GOTO done

LABEL pass
PRINT DDS loopback self-test passed

LABEL done
